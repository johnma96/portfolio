% Texto del enlace de cada curso; cada versión lo define en su idioma
\newcommand{\cvcertlabel}{Certificado}

% Curso o certificación en formato compacto:
% \cvcert{nombre}{emisor}{fecha}{url del certificado (opcional)}
\newcommand{\cvcert}[4]{%
  {\color{emphasis}\hyphenpenalty=10000 #1\par}\nopagebreak
  {\small\textbf{\color{accent}#2}\enspace\textperiodcentered\enspace#3%
    \ifstrequal{#4}{}{}{\enspace\textperiodcentered\enspace\href{#4}{\textcolor{accent}{\faLink}~\cvcertlabel}}\par}
  \smallskip
}

% Grupo de habilidades como palabras clave (legible para ATS):
% \cvskillgroup{grupo}{herramientas separadas por comas}
\newcommand{\cvskillgroup}[2]{%
  {\small\textbf{\color{emphasis}#1:}~#2\par}\smallskip
}
